-- .Models.Model.mAssignOTA

param WO : object

var OTAProbability : real
var WiFiProbability : real

-- Do not reassign OTA if this WO has already passed through the assignment method.
if WO.OTAMethod /= ""
    return
end

-- Default: no OTA required.
WO.OTARequired := false
WO.OTAAffectedQuantity := 0
WO.OTAMethod := "None"
WO.OTADuration := 0:00:00

-- Read provisional probabilities from OTAParameters.
OTAProbability := OTAParameters[2, 5]
WiFiProbability := OTAParameters[2, 2]

-- Determine whether this WO requires OTA.
if z_uniform(1, 0, 1) < OTAProbability

    WO.OTARequired := true

    -- Sample the number of affected devices.
    WO.OTAAffectedQuantity := mGetOTAAffectedQuantity(WO.WorkOrderQuantity)

    -- Select the OTA method.
if z_uniform(1, 0, 1) < WiFiProbability

    WO.OTAMethod := "WiFi"

    -- Provisional WiFi OTA duration in minutes.
    var WiFiMinutes : real
		WiFiMinutes := z_triangle(1, 45, 30, 60)

    WO.OTADuration := WiFiMinutes * 1:00

else

    WO.OTAMethod := "USB"

    -- USB updates are performed sequentially.
    WO.OTADuration := WO.OTAAffectedQuantity * 6:00

end

end
