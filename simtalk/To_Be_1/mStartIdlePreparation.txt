-- .Models.Model.mStartIdlePreparation
param DSO : integer
-> boolean
var Batch : object

-- Check whether this DSO is currently eligible to begin advance preparation.
if mCanDSOStartPreparation(DSO) = false
    return false
end

-- Select group, determine batch quantity, reserve inventory capacity and create the batch.
Batch := mCreatePreparationBatch(DSO)

if Batch = void
    return false
end

-- Mark the DSO as occupied by preparation.
DSO_State[2, DSO] := true

print "Idle DSO starting preparation."
print "DSO: ", DSO
print "Group: ", Batch.Group
print "Quantity: ", Batch.BatchQuantity

-- Start the preparation route.
mStartPreparationBatch(Batch)

return true
