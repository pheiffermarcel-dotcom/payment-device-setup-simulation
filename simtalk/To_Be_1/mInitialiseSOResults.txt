-- .Models.Model.mInitialiseSOResults

SO_Results.deleteContents

for var i := 1 to WO_History.yDim

    var CurrentSO : string
    var ResultRow : integer

    CurrentSO := WO_History[1, i]
    ResultRow := 0

    -- Find the existing SO.
    for var j := 1 to SO_Results.yDim

        if SO_Results[1, j] = CurrentSO
            ResultRow := j
            exitloop
        end

    next

    -- Create a new SO record if necessary.
    if ResultRow = 0

        ResultRow := SO_Results.yDim + 1

        SO_Results[1, ResultRow] := CurrentSO
        SO_Results[2, ResultRow] := 0
        SO_Results[3, ResultRow] := 0

        -- First WO creation timestamp.
        SO_Results[4, ResultRow] := WO_History[6, i]

        SO_Results[7, ResultRow] := 0

    end

    -- Determine the earliest WO creation timestamp belonging to this SO.
    if WO_History[6, i] < SO_Results[4, ResultRow]

        SO_Results[4, ResultRow] := WO_History[6, i]

    end

    -- Count the WOs belonging to this SO.
    SO_Results[2, ResultRow] := SO_Results[2, ResultRow] + 1

    -- Accumulate the total device quantity.
    SO_Results[7, ResultRow] :=
        SO_Results[7, ResultRow] + WO_History[7, i]

next
