var TotalSOSLT : time
var CompletedSOCount : integer

TotalSOSLT := 0:00:00
CompletedSOCount := 0

for var i := 1 to SO_Results.yDim

    if SO_Results[3, i] = SO_Results[2, i]

        TotalSOSLT := TotalSOSLT + SO_Results[6, i]
        CompletedSOCount := CompletedSOCount + 1

    end

next

if CompletedSOCount > 0
    MeanSOSLT := TotalSOSLT / CompletedSOCount
else
    MeanSOSLT := 0:00:00
end

-- Calculate mean historical WO utilisation.
-- This uses the same utilisation definition as the As-Is model: historical WO busy time / HistoricalElapsedTime.
var TotalWorkOrderUtilisation : real

    TotalWorkOrderUtilisation :=
        TotalWorkOrderUtilisation + DSO_Results[6, i]
next

if NumberOfDSOs > 0
    MeanWorkOrderDSOUtilisation :=
        TotalWorkOrderUtilisation / NumberOfDSOs
else
    MeanWorkOrderDSOUtilisation := 0.0
end
