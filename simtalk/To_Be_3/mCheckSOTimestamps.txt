-- .Models.Model.mCheckSOTimestamps

var IssueCount : integer
IssueCount := 0

for var i := 1 to SO_Results.yDim

    -- Verify that the earliest WO creation timestamp does not occur after SO setup completion.
    if SO_Results[4, i] > SO_Results[5, i] or

       -- Verify that SOSLT is calculated from earliest WO creation to final WO completion.
       SO_Results[6, i] /=
           SO_Results[5, i] - SO_Results[4, i]

        IssueCount := IssueCount + 1

        print "TIMESTAMP ISSUE: ", SO_Results[1, i]

    end

next

print "Sales Orders checked: ", SO_Results.yDim
print "Sales Orders with timestamp or lead-time issues: ", IssueCount
